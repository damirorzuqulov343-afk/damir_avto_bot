import requests
import time

BOT_TOKEN = os.environ.get("BOT_TOKEN", "BU YERGA TOKEN")
# Agar BOT_TOKEN muhitdan olinmasa, pastga qo'lda yozib qo'yish mumkin

AVTO_JAVOB = "Salom! Hozir bandman, birozdan keyin albatta javob beraman. Rahmat! 🙏"

API = f"https://api.telegram.org/bot{BOT_TOKEN}"

javob_berilgan = set()

def send_business_reply(chat_id, business_connection_id):
    try:
        r = requests.post(f"{API}/sendMessage", json={
            "chat_id": chat_id,
            "text": AVTO_JAVOB,
            "business_connection_id": business_connection_id
        }, timeout=10)
        data = r.json()
        print(f"Yuborildi {chat_id} ga: {data.get('ok')}")
        return data.get("ok", False)
    except Exception as e:
        print(f"Yuborishda xatolik: {e}")
        return False

def main():
    print("Doimiy bot ishga tushdi...")
    offset = 0
    while True:
        try:
            resp = requests.get(f"{API}/getUpdates", params={
                "offset": offset,
                "timeout": 30,
                "allowed_updates": ["business_connection", "business_message", "message"]
            }, timeout=35)
            data = resp.json()
            if not data.get("ok"):
                time.sleep(2)
                continue

            for upd in data.get("result", []):
                offset = upd["update_id"] + 1

                # Business ulandi
                if "business_connection" in upd:
                    user = upd["business_connection"].get("user", {})
                    print(f"Business ulandi: {user.get('first_name')}")

                # Business xabar
                if "business_message" in upd:
                    msg = upd["business_message"]
                    chat_id = msg["chat"]["id"]
                    b_conn_id = msg.get("business_connection_id")
                    # Bot o'zi yozgan bo'lsa o'tkazib yuborish
                    if msg.get("from", {}).get("is_bot"):
                        continue
                    # Har bir chatga har safar javob (test uchun)
                    # Agar 1 soatda 1 marta bo'lishini xohlasang pastdagi 2 qatorni yoq
                    # if chat_id in javob_berilgan:
                    #     continue
                    print(f"Xabar keldi {chat_id} dan -> avto javob yuborilyapti")
                    if b_conn_id:
                        ok = send_business_reply(chat_id, b_conn_id)
                        if ok:
                            javob_berilgan.add(chat_id)

        except Exception as e:
            print(f"Loop xatolik: {e}")
            time.sleep(5)

if __name__ == "__main__":
    import os
    main()
